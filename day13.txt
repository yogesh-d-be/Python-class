print("Select your language: English")

balance = 10000

try:
    account = input("Select your account: S/C ")
    if(account.lower() == "s"):
        print("Savings account")
    elif(account.lower() == "c"):
        print("Current account")
    else:
        raise Exception("Please select valid account type")

    amount = int(input("Enter your amount: "))

    if(amount > balance):
        raise ValueError("Insufficent balance")
    elif(amount < 100):
        raise ValueError("Please enter valid amount value")

    pin = input("Enter your pin: ")

    if(pin != "1111"):
        raise ValueError("Please enter valid pin number")
    
    print("Please collect your cash: ", amount)

except Exception as err:
    print(err)
else:
    print("Amount collected successfully")
finally:
    print("Please take your card. Thank you!")
